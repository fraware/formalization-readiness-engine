\section{Graph sketches (metadata-only release)}

\begin{theorem}[Sketch: tree edge bound]
Every finite tree satisfies $|E| \le |V| - 1$.
\end{theorem}

\begin{lemma}[Sketch: leaf existence]
Every finite nontrivial tree has a leaf.
\end{lemma}

\begin{definition}[Sketch: spanning tree]
A spanning tree of a connected graph is a subgraph that is a tree and includes all vertices.
\end{definition}
